% elle agrandit la case et le tableau descend d'autant.
% \IfFileExists ET NON UN \includegraphics NU, et c'est ce qui permet de garder
% les griffes HORS du depot. Elles ne sont pas versionnees : le depot GitHub est
% public, et une signature qu'on y pousse reste dans l'historique meme effacee,
% recuperable et reutilisable par n'importe qui. Sans ce garde-fou, un clone
% neuf ne compilerait plus, faute de trouver le fichier ; avec lui, la case sort
% simplement vide et le document se construit partout.

  \IfFileExists{#1}{\raisebox{-24pt}[0pt][0pt]{\includegraphics[height=30pt,keepaspectratio]{#1}}}{}}
